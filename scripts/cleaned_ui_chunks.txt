    const App = {

    function setStatus(kind, text) {
      el.statusText.textContent = text;
      if (kind === "ok") el.statusDot.style.background = "var(--ok)";
      else if (kind === "error") el.statusDot.style.background = "var(--danger)";
      else el.statusDot.style.background = "rgba(255,255,255,0.25)";
    }

    function showError(message) {
      el.errorMsg.textContent = String(message || "Unknown error");
      el.errorSurface.style.display = "inline-flex";
    }

    function clearErrorSurface() {
      el.errorMsg.textContent = "";
      el.errorSurface.style.display = "none";
    }

    function safeJsonParse(text, contextLabel) {
      try { return JSON.parse(text); }
      catch (e) {
        throw new Error(`Invalid JSON in ${contextLabel}: ${e && e.message ? e.message : e}`);
      }
    }

    async function loadConfigOnce() {
      const resp = await fetch("./config.json", { cache: "no-store" });
      if (!resp.ok) throw new Error(`Failed to load config.json (${resp.status})`);
      const text = await resp.text();
      const cfg = safeJsonParse(text, "web/config.json");
      App.config = cfg;
      App.state.config_loaded = true;
    }

    async function loadDataOnce() {
      const resp = await fetch("../data/data.json", { cache: "no-store" });
      if (!resp.ok) throw new Error(`Failed to load data.json (${resp.status})`);
      const text = await resp.text();
      const data = safeJsonParse(text, "data/data.json");
      App.frozen_data = Object.freeze(data);
      App.state.data_loaded = true;
    }

    function setActiveNav(viewKey) {

    function isoDayKeyFromDate(d) {
      const y = d.getFullYear();
      const m = pad2(d.getMonth() + 1);
      const day = pad2(d.getDate());
      return `${y}-${m}-${day}`;

    function parseISODateOnly(s) {
      if (!s || typeof s !== "string") return null;
      // Accept YYYY-MM-DD; reject anything else deterministically
      const m = s.match(/^(\d{4})-(\d{2})-(\d{2})$/);
      if (!m) return null;
      const y = Number(m[1]), mo = Number(m[2]), da = Number(m[3]);
      if (!Number.isFinite(y) || !Number.isFinite(mo) || !Number.isFinite(da)) return null;
      const d = new Date(Date.UTC(y, mo - 1, da));
      // Validate round-trip (invalid dates collapse)
      if (d.getUTCFullYear() !== y || (d.getUTCMonth() + 1) !== mo || d.getUTCDate() !== da) return null;
      // Return local date with same Y-M-D for consistent calendar placement
      return new Date(y, mo - 1, da);
    }

    function firstDayOffsetMonday(year, monthIndex) {
      // Returns 0..6 offset where 0 is Monday
      const d = new Date(year, monthIndex, 1);
      const jsDow = d.getDay(); // 0=Sun..6=Sat
      // Convert to Monday-first: Mon=0..Sun=6
      return (jsDow + 6) % 7;
    }

    function daysInMonth(year, monthIndex) {
      return new Date(year, monthIndex + 1, 0).getDate();
    }

    function normalizeText(s, fallback) {
      const t = (typeof s === "string") ? s.trim() : "";

    }

      const hasPopups = App.popup_stack.length &gt; 0;
      el.popupOverlay.style.display = hasPopups ? "block" : "none";
      el.popupContainer.style.display = hasPopups ? "block" : "none";
      el.popupOverlay.setAttribute("aria-hidden", hasPopups ? "false" : "true");

      el.popupContainer.innerHTML = "";
      if (!hasPopups) return;

      for (let i = 0; i &lt; App.popup_stack.length; i++) {
        const p = App.popup_stack[i];

        const popup = document.createElement("div");
        popup.className = "popup";
        popup.setAttribute("role", "dialog");
        popup.setAttribute("aria-modal", "true");
        popup.setAttribute("data-popup-index", String(i));
        popup.style.zIndex = String(90 + i);

        const header = document.createElement("div");
        header.className = "popup-header";

        const t = document.createElement("div");
        t.className = "title";
        t.textContent = p && p.title ? String(p.title) : "Popup";

        const spacer = document.createElement("div");
        spacer.className = "spacer";

        const closeBtn = document.createElement("button");
        closeBtn.type = "button";
        closeBtn.className = "btn";
        closeBtn.textContent = "Close";
        closeBtn.addEventListener("click", () =&gt; closeTopPopup());

        header.appendChild(t);
        header.appendChild(spacer);
        header.appendChild(closeBtn);

        const body = document.createElement("div");
        body.className = "popup-body";
        body.tabIndex = 0;

        const bodyText = document.createElement("div");
        bodyText.className = "hint";
        bodyText.textContent = p && p.body ? String(p.body) : "Popup body container ready.";
        body.appendChild(bodyText);

        popup.appendChild(header);
        popup.appendChild(body);

        el.popupContainer.appendChild(popup);
      }

      const topPopup = el.popupContainer.querySelector(".popup:last-child .popup-body");
      if (topPopup) topPopup.focus({ preventScroll: true });
    }

    function openPopup(payload) {
      App.popup_stack.push(payload || { title: "Popup", body: "" });
      syncPopupDom();
    }

    function closeTopPopup() {
      if (App.popup_stack.length === 0) return;
      App.popup_stack.pop();
      syncPopupDom();
      if (App.popup_stack.length === 0) {
        el.contentRoot.focus({ preventScroll: true });
      }
    }

    function closeAllPopups() {
      if (App.popup_stack.length === 0) return;
      App.popup_stack = [];

      el.contentRoot.focus({ preventScroll: true });
    }

    function parseDate(dateStr) {
      if (!dateStr) return null;
      const d = new Date(dateStr);
      return isNaN(d.getTime()) ? null : d;
    }

    function formatDate(dateStr) {
      const d = parseDate(dateStr);
      if (!d) return "Unknown date";
      return d.toLocaleDateString(undefined, {
        year: "numeric",
        month: "short",
        day: "numeric"
      });
    }

    function safeText(value, fallback = "Unknown") {
      if (value === null || value === undefined) return fallback;
      if (typeof value === "string" && value.trim() === "") return fallback;
      return value;
    }

    function createEl(tag, className, text) {
      const el = document.createElement(tag);
      if (className) el.className = className;
      if (text !== undefined && text !== null) el.textContent = text;
      return el;
    }

    function applyFiltersAndSort() {
      const { allShows, currentStatusFilter, currentSort } = state;
      let list = [...allShows];

      // Filter by status
      if (currentStatusFilter !== "ALL") {
        list = list.filter((show) =&gt; String(show.status || "").toUpperCase() === currentStatusFilter);
      }

      // Sort
      list.sort((a, b) =&gt; compareShows(a, b, currentSort));

      state.filteredShows = list;
      renderWatchlist();
    }

    function compareShows(a, b, sortKey) {
      const titleA = (a.title || a.name || "").toLowerCase();
      const titleB = (b.title || b.name || "").toLowerCase();

      const lastSeasonA = parseDate(a.last_season_air_date || a.lastSeasonAirDate);
      const lastSeasonB = parseDate(b.last_season_air_date || b.lastSeasonAirDate);

      const lastEpisodeA = parseDate(a.last_episode_air_date || a.lastEpisodeAirDate);
      const lastEpisodeB = parseDate(b.last_episode_air_date || b.lastEpisodeAirDate);

      switch (sortKey) {
        case "title-desc":
          if (titleA &lt; titleB) return 1;
          if (titleA &gt; titleB) return -1;
          return 0;
        case "last-season-desc":
          if (lastSeasonA && lastSeasonB) return lastSeasonB - lastSeasonA;
          if (lastSeasonA) return -1;
          if (lastSeasonB) return 1;
          return 0;
        case "last-season-asc":
          if (lastSeasonA && lastSeasonB) return lastSeasonA - lastSeasonB;
          if (lastSeasonA) return 1;
          if (lastSeasonB) return -1;
          return 0;
        case "last-episode-desc":
          if (lastEpisodeA && lastEpisodeB) return lastEpisodeB - lastEpisodeA;
          if (lastEpisodeA) return -1;
          if (lastEpisodeB) return 1;
          return 0;
        case "last-episode-asc":
          if (lastEpisodeA && lastEpisodeB) return lastEpisodeA - lastEpisodeB;
          if (lastEpisodeA) return 1;
          if (lastEpisodeB) return -1;
          return 0;
        case "title-asc":
        default:
          if (titleA &lt; titleB) return -1;
          if (titleA &gt; titleB) return 1;
          return 0;

    function populateEpisodesForSeason(show, seasonNumber, episodesList) {
      if (!episodesList) return;

      if (!Array.isArray(show.episodes)) {
        const empty = createEl("div", "section-subtitle", "No episode data available for this show.");
        episodesList.appendChild(empty);

      const episodes = show.episodes.filter((ep) =&gt; ep && ep.season_number === seasonNumber);
      episodes.sort((a, b) =&gt; (a.episode_number || 0) - (b.episode_number || 0));

      if (episodes.length === 0) {
        const empty = createEl(
          "div",
          "section-subtitle",
          `No episodes found for season ${seasonNumber}.`
        );
        episodesList.appendChild(empty);
        return;
      }

      episodes.forEach((ep) =&gt; {
        const card = renderEpisodeCard(ep);
        episodesList.appendChild(card);
      });
    }
